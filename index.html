It is " + currentPokemon.name.toUpperCase() + "!"; document.getElementById("score").innerText = "Score: " + score; return; } try { const response = await fetch(`https://pokeapi.co/api/v2/pokemon-species/${guess}`); const guessData = await response.json(); const guessGen = getGeneration(guessData.id); if (guessGen < currentPokemon.gen) { document.getElementById("result").innerText = "Wrong!
